8) calculate mode (most frequent element in a column)

first get the count of each element , assign the rank and take the first one which has first rank (order by desc)

create table mode 
(
id int
);

insert into mode values (1),(2),(2),(3),(3),(3),(3),(4),(5);

insert into mode values (4),(4),(4);
select *from mode

with freq_cte as (
select id, count(1) as freq from mode group by id
),rank_cte as (
select *, rank() over(order by freq desc) as rn
from freq_cte
)
select *from rank_cte where rn=1

10) rank only dupes

create table list (id varchar(5));
insert into list values ('a');
insert into list values ('a');
insert into list values ('b');
insert into list values ('c');
insert into list values ('c');
insert into list values ('c');
insert into list values ('d');
insert into list values ('d');
insert into list values ('e');

select *from list;

with dupe_ids as (
select id from list group by id having count(1) >1
), rank_dupes as (
select *,
rank() over(order by id asc) as rn
from dupe_ids
)
select l.id, 'DUP'+ cast(rd.rn as varchar(2)) from list l left join rank_dupes rd on l.id = rd.id
