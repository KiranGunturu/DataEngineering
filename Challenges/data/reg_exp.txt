[abc] = a , b or c
[^abc] = any character except a, b, c
[a-z] = a to z
[A-Z] = A to Z
[a-z A-Z] = a to z , A to Z
[0-9]= 0 to 9

[ ] ? = Occurs 0 or 1 times
[ ] + = Occurs 1 or more times
[ ] * = Occurs 0 or more times
[ ] {n} = Occurs n times
[ ] {n, } = occurs n or more times
[ ] {y,z] = occurs atleast y times but less than z times

\d = [0-9]
\D = [^0-9]
\w = [a-z A-Z 0-9]
\W = [^w]
