from pyspark import SparkContext
from pyspark.sql import SparkSession

spark = SparkSession \
        .builder \
        .appName("Hello world") \
        .master("local[2]") \
        .getOrCreate()
        
print(spark.version)

orderdf = spark.read \
			.option("header","true") \
			.optin("inferSchema","true") \
			.csv("/path/orders.csv")

orderdf.show()

orderdf.printSchema()

there are 3 actions here


read the data and inferSchema

3rd is for show()
